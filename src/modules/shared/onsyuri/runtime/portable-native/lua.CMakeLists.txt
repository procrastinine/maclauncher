cmake_minimum_required(VERSION 3.16)
project(lua LANGUAGES C)
set(CMAKE_C_STANDARD 99)
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS src/*.c)
list(REMOVE_ITEM LUA_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/lua.c" "${CMAKE_CURRENT_SOURCE_DIR}/src/luac.c")
add_library(lua SHARED ${LUA_SOURCES})
target_compile_definitions(lua PRIVATE LUA_USE_MACOSX)
target_include_directories(lua PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>" "$<INSTALL_INTERFACE:include>")
set_target_properties(lua PROPERTIES VERSION 5.4.9 SOVERSION 5.4)
install(TARGETS lua LIBRARY DESTINATION lib)
install(FILES src/lua.h src/luaconf.h src/lauxlib.h src/lualib.h src/lua.hpp DESTINATION include)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/lua.pc" "prefix=${CMAKE_INSTALL_PREFIX}\nlibdir=\${prefix}/lib\nincludedir=\${prefix}/include\nName: Lua\nDescription: Lua 5.4\nVersion: 5.4.9\nLibs: -L\${libdir} -llua\nCflags: -I\${includedir}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/lua.pc" DESTINATION lib/pkgconfig)
