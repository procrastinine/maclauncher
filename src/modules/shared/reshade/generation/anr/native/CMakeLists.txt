cmake_minimum_required(VERSION 3.20)
set(CMAKE_OSX_DEPLOYMENT_TARGET "26.0" CACHE STRING "Minimum macOS")
project(anr_provider LANGUAGES CXX OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD 17)
add_compile_options(-fobjc-arc "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=ANRProvider" "-ffile-prefix-map=${CMAKE_CURRENT_BINARY_DIR}=ANRBuild")
if(NOT ANR_SHARED_INCLUDE OR NOT ANR_TEMPORAL_INCLUDE OR NOT ANR_CONVERTER_INCLUDE)
 message(FATAL_ERROR "ANR requires shared ABI, temporal ABI and pinned converter header paths")
endif()
include_directories("${ANR_SHARED_INCLUDE}" "${ANR_TEMPORAL_INCLUDE}" "${ANR_CONVERTER_INCLUDE}")
add_library(maclauncher-anr-generation SHARED backend.mm anr_runtime.mm)
set_target_properties(maclauncher-anr-generation PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR TRUE)
add_executable(anr-convert-dxil convert_dxil.mm)
add_executable(anr-verify verify.mm anr_runtime.mm)
foreach(target maclauncher-anr-generation anr-convert-dxil anr-verify)
 target_link_libraries(${target} PRIVATE "-framework Foundation" "-framework Metal")
endforeach()
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/helpers.metallib"
 COMMAND xcrun -sdk macosx metal -std=metal3.0 -O2 -c "${CMAKE_CURRENT_SOURCE_DIR}/helpers.metal" -o "${CMAKE_CURRENT_BINARY_DIR}/helpers.air"
 COMMAND xcrun -sdk macosx metallib "${CMAKE_CURRENT_BINARY_DIR}/helpers.air" -o "${CMAKE_CURRENT_BINARY_DIR}/helpers.metallib"
 DEPENDS helpers.metal VERBATIM)
add_custom_target(anr-helpers ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/helpers.metallib")
install(TARGETS maclauncher-anr-generation anr-convert-dxil anr-verify LIBRARY DESTINATION . RUNTIME DESTINATION .)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/helpers.metallib" DESTINATION .)
