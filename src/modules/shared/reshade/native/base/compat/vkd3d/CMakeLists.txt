cmake_minimum_required(VERSION 3.20)
set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS version")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
 project(ls1_dxbc_translator C)
endif()
set(VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/vkd3d")
set(SPIRV "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/reshade/deps/spirv/include")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/config.h" "#define HAVE_XLOCALE_H 1\n#define HAVE_STRTOF_L 1\n#define HAVE_PTHREAD_H 1\n#define HAVE_PTHREAD_THREADID_NP 1\n#define HAVE_PTHREAD_SETNAME_NP_1 1\n#define HAVE_BUILTIN_CLZ 1\n#define HAVE_BUILTIN_CTZ 1\n#define HAVE_BUILTIN_POPCOUNT 1\n#define HAVE_BUILTIN_ADD_OVERFLOW 1\n#define HAVE_SYNC_ADD_AND_FETCH 1\n#define HAVE_SYNC_BOOL_COMPARE_AND_SWAP 1\n#define HAVE_ATOMIC_EXCHANGE_N 1\n#define HAVE_SPIRV_UNIFIED1_SPIRV_H 1\n#define HAVE_SPIRV_UNIFIED1_GLSL_STD_450_H 1\n#define PACKAGE_VERSION \"2.1\"\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/vkd3d_version.h" "#define VKD3D_VCS_ID \"vkd3d-2.1\"\n")
set(SOURCES dxbc-only.c)
foreach(name checksum d3d_asm d3dbc dxbc dxil glsl ir msl spirv tpf)
 list(APPEND SOURCES "${VENDOR}/libs/vkd3d-shader/${name}.c")
endforeach()
foreach(name debug error memory utf8)
 list(APPEND SOURCES "${VENDOR}/libs/vkd3d-common/${name}.c")
endforeach()
# Do not advertise source languages intentionally excluded from this build.
file(READ "${VENDOR}/libs/vkd3d-shader/vkd3d_shader_main.c" SHADER_MAIN)
foreach(kind HLSL FX TX)
 string(FIND "${SHADER_MAIN}" "        VKD3D_SHADER_SOURCE_${kind}," FOUND_SOURCE)
 if(FOUND_SOURCE LESS 0)
  message(FATAL_ERROR "vkd3d source capabilities changed; review the binary-only build")
 endif()
 string(REPLACE "        VKD3D_SHADER_SOURCE_${kind},\n" "" SHADER_MAIN "${SHADER_MAIN}")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/vkd3d_shader_main.c" "${SHADER_MAIN}")
add_library(vkd3d-shader SHARED ${SOURCES} "${CMAKE_CURRENT_BINARY_DIR}/vkd3d_shader_main.c")
target_include_directories(vkd3d-shader PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/include/private" "${VENDOR}/include" "${VENDOR}/include/private" "${VENDOR}/libs/vkd3d-shader" "${SPIRV}")
target_compile_definitions(vkd3d-shader PRIVATE LIBVKD3D_SHADER_SOURCE VKD3D_DEBUG_ENV=VKD3D_SHADER_DEBUG VKD3D_NO_TRACE_MESSAGES VKD3D_NO_DEBUG_MESSAGES)
target_compile_options(vkd3d-shader PRIVATE -fvisibility=hidden -Wno-ignored-attributes -Wno-unknown-attributes)
