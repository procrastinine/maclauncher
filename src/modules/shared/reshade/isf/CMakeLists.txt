cmake_minimum_required(VERSION 3.20)
project(MacLauncherISF LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OSX_DEPLOYMENT_TARGET 13.0)
set(CMAKE_OSX_ARCHITECTURES "arm64;x86_64" CACHE STRING "Native provider CPUs")
if(NOT ISF_INPUT OR NOT MACFX_INPUT)
  message(FATAL_ERROR "ISF_INPUT and MACFX_INPUT are required")
endif()
file(GLOB ISF_GENERATOR "${ISF_INPUT}/generator/*.cpp")
file(GLOB ISF_SCENE "${ISF_INPUT}/ISFMSLKit/*.mm")
file(GLOB ISF_METAL "${ISF_INPUT}/VVMetalKit/*.m" "${ISF_INPUT}/VVMetalKit/*.mm")
list(FILTER ISF_METAL EXCLUDE REGEX "/(CMVMTL[^/]*|CustomMetalView|VVMTLTextureImage(Rect)?View|CIMTLScene|VVMTLPerspRenderScene|VVMTLOrthoRenderScene|VVMTLRenderScene)\\.(m|mm)$")
file(GLOB ISF_CACHE "${ISF_INPUT}/PINCache/*.m" "${ISF_INPUT}/PINOperation/*.m")
add_library(macfx-isf SHARED ${ISF_GENERATOR} ${ISF_SCENE} ${ISF_METAL} ${ISF_CACHE} provider.mm)
target_compile_definitions(macfx-isf PRIVATE MAC=1)
target_include_directories(macfx-isf PRIVATE "${ISF_INPUT}" "${ISF_INPUT}/generator" "${ISF_INPUT}/ISFMSLKit" "${ISF_INPUT}/VVMetalKit" "${ISF_INPUT}/PINCache" "${ISF_INPUT}/PINOperation" "${MACFX_INPUT}" "${MACFX_INPUT}/base/src")
target_compile_options(macfx-isf PRIVATE -O2 -fvisibility=hidden -fvisibility-inlines-hidden -Wno-deprecated-declarations -Werror=unguarded-availability-new -include "${ISF_INPUT}/prefix.h")
target_compile_options(macfx-isf PRIVATE "-ffile-prefix-map=${ISF_INPUT}=MacLauncherISFInputs" "-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=MacLauncherISFProvider" "-ffile-prefix-map=${MACFX_INPUT}=MacLauncherGraphics")
set_source_files_properties(${ISF_SCENE} ${ISF_METAL} ${ISF_CACHE} provider.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
target_link_options(macfx-isf PRIVATE -Wl,-undefined,dynamic_lookup -Wl,-rpath,@loader_path)
target_link_libraries(macfx-isf PRIVATE "${ISF_INPUT}/libGLSLangValidatorLib.dylib" "${ISF_INPUT}/libSPIRVCrossLib.dylib" "-framework Foundation" "-framework AppKit" "-framework Metal" "-framework MetalKit" "-framework CoreVideo" "-framework CoreImage" "-framework QuartzCore" "-framework IOSurface")
set_target_properties(macfx-isf PROPERTIES OUTPUT_NAME macfx-isf INSTALL_NAME_DIR @rpath BUILD_WITH_INSTALL_NAME_DIR YES BUILD_WITH_INSTALL_RPATH YES INSTALL_RPATH @loader_path)
